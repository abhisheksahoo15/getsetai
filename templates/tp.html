# GetSetAI quick start (FastAPI)
from fastapi import FastAPI
from getsetai import Models
app = FastAPI(title="GetSetAI")
# Search catalog
CATALOG = Models.list()  # ['llama-3.1','whisper-medium','yolov8',...]
@app.get("/classify")
def classify(text: str):
    verdict = Models.use("sentiment-mini").predict(text)
    return {"verdict": verdict}
